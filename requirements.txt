numpy
# scikit-learn: preprocessing, models, pipelines and evaluation metrics.
scikit-learn

# ---- Configuration ----
# pyyaml: reads the settings in config/default.yaml (imported as `yaml`).
pyyaml

# ---- Testing ----
# pytest: finds and runs the tests in tests/ (`make test`).
pytest
