# Python packages this project needs. Install them into the virtual
# environment with `make venv` (which runs pip install -r requirements.txt).
# One package per line. A line can pin a version (for example pandas==2.2.3);
# here we leave versions unpinned so pip installs the latest release.
# Lines that start with # are comments and pip ignores them.

# ---- Data handling and machine learning ----
# pandas: reads the CSV file and holds the data in DataFrames.
pandas
# numpy: fast numerical arrays; pandas and scikit-learn are built on it.
numpy
# scikit-learn: preprocessing, models, pipelines and evaluation metrics.
scikit-learn

# ---- Configuration ----
# pyyaml: reads the settings in config/default.yaml (imported as `yaml`).
pyyaml

# ---- Experiment tracking ----
# mlflow: records parameters, metrics and models for every training run,
# and provides the web UI (`make mlflow-ui`) for comparing them.
mlflow
# skops: a safe file format for scikit-learn models. MLflow saves sklearn
# models with skops so that loading a model does not run arbitrary code,
# which loading a pickle file can do.
skops

# ---- Testing ----
# pytest: finds and runs the tests in tests/ (`make test`).
pytest

# ---- Data version control ----
# dvc: versions large data and model files outside Git (`dvc add`, push, pull).
dvc
# dvc-s3: lets DVC talk to S3-compatible storage. DagsHub's DVC remote uses
# the S3 API, so DVC needs this plug-in to push and pull.
dvc-s3
